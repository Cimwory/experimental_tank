        document.addEventListener('alpine:init', () => {
            Alpine.data('tankDashboard', () => ({
                // Threshold configuration
                thresholds: {
                    low: 20,
                    high: 70
                },

                // Dummy Initial Data
                tanks: [
                    { id: 1, name: 'RO Tank', capacity: 500, current_volume: 325, water_level: 65, status: 'MID', temperature: 26.4, pump_status: true, valve_status: true, last_update: '14:02:31', connection: 'ONLINE' },
                    { id: 2, name: 'Nutrient Tank A', capacity: 1000, current_volume: 850, water_level: 85, status: 'HIGH', temperature: 22.1, pump_status: false, valve_status: false, last_update: '14:02:30', connection: 'ONLINE' },
                    { id: 3, name: 'Mixing Tank', capacity: 200, current_volume: 30, water_level: 15, status: 'LOW', temperature: 28.5, pump_status: true, valve_status: false, last_update: '14:02:29', connection: 'ONLINE' },
                    { id: 4, name: 'Irrigation Tank', capacity: 2000, current_volume: 0, water_level: 0, status: 'OFFLINE', temperature: null, pump_status: false, valve_status: false, last_update: '10:00:00', connection: 'OFFLINE' }
                ],
                
                selectedTankId: null,
                isSimulating: false,
                simulationInterval: null,

                // Computed getter for selected tank
                get selectedTank() {
                    if(this.selectedTankId === null) return null;
                    return this.tanks.find(t => t.id === this.selectedTankId);
                },

                // Actions
                selectTank(id) {
                    this.selectedTankId = id;
                },
                
                closeDetail() {
                    this.selectedTankId = null;
                },

                // Core function to update data from API/WebSocket
                updateTankData(id, newData) {
                    const index = this.tanks.findIndex(t => t.id === id);
                    if(index !== -1) {
                        // Merge old and new data
                        let updatedTank = { ...this.tanks[index], ...newData };
                        
                        // Auto-recalculate status based on thresholds if level changes
                        if (newData.water_level !== undefined) {
                            if (updatedTank.connection === 'OFFLINE') {
                                updatedTank.status = 'OFFLINE';
                            } else if (newData.water_level <= this.thresholds.low) {
                                updatedTank.status = 'LOW';
                            } else if (newData.water_level >= this.thresholds.high) {
                                updatedTank.status = 'HIGH';
                            } else {
                                updatedTank.status = 'MID';
                            }
                            
                            // Auto-calculate volume if capacity is known
                            updatedTank.current_volume = Math.round((newData.water_level / 100) * updatedTank.capacity);
                        }

                        // Trigger UI update
                        this.tanks[index] = updatedTank;
                    }
                },

                // Real-time Dummy Simulator
                toggleRealtimeSimulation() {
                    this.isSimulating = !this.isSimulating;
                    if (this.isSimulating) {
                        this.simulationInterval = setInterval(() => {
                            this.tanks.forEach(tank => {
                                // Skip offline tanks
                                if(tank.connection === 'OFFLINE' || tank.status === 'ERROR') return;

                                // Generate random fluctuation (-3 to +3 percent)
                                let fluctuation = (Math.random() * 6) - 3;
                                
                                // Specific logic: if pump is ON, it should generally increase
                                if (tank.pump_status) fluctuation = Math.random() * 5; 
                                
                                let newLevel = tank.water_level + fluctuation;
                                
                                // Clamp between 0 and 100
                                if(newLevel > 100) newLevel = 100;
                                if(newLevel < 0) newLevel = 0;

                                // Randomly toggle pump/valve for demo purposes (10% chance)
                                let newPump = tank.pump_status;
                                let newValve = tank.valve_status;
                                if(Math.random() < 0.1) newPump = !newPump;
                                if(Math.random() < 0.1) newValve = !newValve;

                                this.updateTankData(tank.id, {
                                    water_level: newLevel,
                                    pump_status: newPump,
                                    valve_status: newValve,
                                    last_update: new Date().toLocaleTimeString('id-ID', { hour12: false })
                                });
                            });
                        }, 2500); // Update every 2.5 seconds
                    } else {
                        clearInterval(this.simulationInterval);
                    }
                },

                // --- Visual Helper Functions ---
                
                getWaterColor(status) {
                    switch(status) {

                        case 'MID': return 'bg-gradient-to-t from-blue-600 to-blue-400';

                        case 'ERROR': return 'bg-red-700 text-white ring-1 ring-red-300';
                        case 'OFFLINE': return 'bg-gray-700 text-gray-200 ring-1 ring-gray-400';

                    }
                },

                        case 'ERROR': return 'bg-red-700';
                        case 'OFFLINE': return 'bg-gray-400';
                        default: return 'bg-blue-500'; // MID

Saya telah menyatukan seluruh program menjadi satu file HTML siap jalan (`dashboard-tandon.html`) sesuai dengan praktik stack **TALL** (Tailwind, Alpine.js, Laravel, Livewire/Blade). 

Berikut penjelasan struktur dan fiturnya:

1. **Komponen Modular**: Kode di atas sudah diberikan struktur komentar yang rapi. Di Laravel nanti, Anda bisa memindahkan template di dalam `template x-for="tank in tanks"` menjadi file komponen blade misal: `resources/views/components/tank-card.blade.php`, dan panel laci di samping menjadi `resources/views/components/tank-drawer.blade.php`.
2. **Animasi Level Air & Gelombang (Wave)**: Gelombang dibuat menggunakan CSS `background-image` berbasis data URI (SVG) dan direkatkan secara absolute di ujung atas komponen air. Tinggi air akan bergerak secara *smooth* menggunakan transisi CSS `transition-all duration-1000 ease-in-out`.
3. **Interaktivitas (Hover, Click, Aksesibilitas)**: Tandon bereaksi saat di-_hover_ (`scale-102`) dan bisa di-_click_. Komponen tandon menggunakan atribut `tabindex` dan `@keydown` sehingga detail bisa dibuka menggunakan spasi atau enter jika dinavigasi pakai keyboard (sangat baik untuk *accessibility*).
4. **Simulator Real-Time**: Saya telah menyiapkan fungsi internal Alpine bernama `updateTankData(id, data)` yang sudah siap menampung *payload API* dan MQTT. Untuk pengujian UI ini, saya buat tombol **"Start Simulator"** di sudut kanan atas agar Anda bisa melihat langsung interaksi transisi halus tinggi air dan pergantian warna (LOW/MID/HIGH) jika nilai data berubah.
5. **Drawer Detail Panel**: Panel sengaja tidak memakai modal biasa agar lebih modern (dashboard industri), melainkan muncul dari sebelah kanan secara animasi (Slide-over drawer). Layout juga dijaga untuk tidak bergeser sama .
